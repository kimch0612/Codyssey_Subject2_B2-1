# 시연용 임시 저장 폴더와 실행 함수
DEMO_DIR=$(mktemp -d)
budget() { python3 -B -m budget_app --data-dir "$DEMO_DIR/data" "$@"; }

# 도움말
budget --help

# 카테고리 등록·조회 (등록할 이름은 실행 후 입력)
budget category add
budget category list

# 거래 추가 (실행 후 질문에 따라 입력)
budget add

# 목록·검색
budget list --limit 3
budget search --from 2026-09-01 --to 2026-09-30 --category food --type expense --q 점심 --tag meal

# 거래 수정·삭제 (ID는 실제 거래 ID로 변경)
budget update --id TX-000001 --amount 15000 --memo "점심" --tags "meal,lunch"
budget delete --id TX-000001

# 카테고리 삭제 (삭제할 이름은 실행 후 입력)
budget category remove

# 예산·월별 요약
budget budget set --month 2026-09 --amount 140000
budget summary --month 2026-09 --top 3

# CSV 내보내기·가져오기 (가져올 거래의 카테고리는 미리 등록)
budget export --out "$DEMO_DIR/september.csv" --month 2026-09
budget import --from "$DEMO_DIR/september.csv"

# 직전 명령의 종료 코드
echo $?
